cmake_minimum_required(VERSION 3.20)
project(mcp_executor_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(executor_tests executor_tests.cpp ../src/main_thread_executor.cpp)
target_include_directories(executor_tests PRIVATE ../include)
target_compile_definitions(executor_tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(executor_tests PRIVATE user32)
add_executable(health_tests health_tests.cpp ../src/main_thread_executor.cpp ../src/bridge_health.cpp)
target_include_directories(health_tests PRIVATE ../include)
target_compile_definitions(health_tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(health_tests PRIVATE user32)
enable_testing()
add_test(NAME executor_regressions COMMAND executor_tests)
add_test(NAME health_bookkeeping COMMAND health_tests)
